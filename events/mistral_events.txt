# Mistral Responds to Vale's Decision - Supports Talks
country_event = {
	id = MIS_mistral_supports_talks
	title = "Mistral Supports Vale's Talks"
	desc = "Despite past tensions, Vale's decision to pursue diplomacy has been met with approval in Mistral. This may open the door to deeper cooperation, though some remain wary."

	picture = GFX_report_event_generic_diplomacy

	is_triggered_only = yes
	trigger = {
		# This event is triggered if Vale chose to pursue talks
		has_focus = VAL_open_talks_with_mistral
	}

	option = {
		name = "This is a step in the right direction."
		# Effects for Mistral agreeing with Vale's approach
		add_political_power = 50

		improve_relations_with_Vale = 1.0
	}
}
