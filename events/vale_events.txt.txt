# Vale Talks with Mistral
country_event = {
	id = VAL_talks_with_mistral
	title = "Diplomatic Strain: Vale & Mistral"
	desc = "Despite historical disputes and ongoing tensions, the kingdom of Vale initiates diplomatic talks with Mistral. The road ahead remains uncertain—should we tread carefully to ease Mistral's concerns, seek a balanced compromise, or take a firmer stance?"

	picture = GFX_report_event_generic_diplomacy

	is_triggered_only = yes
	trigger = {
		# Condition to fire the event
		has_focus = VAL_open_talks_with_mistral
	}

	option = {
		name = "Appease Mistral to avoid escalation."
		# Prioritizing diplomacy at the cost of concessions
		add_political_power = -40
		improve_relations_with_Mistral = 1.0
		decrease_stability = 0.20
		trigger_event = MIS_mistral_supports_talks

		complete_focus = VAL_placate_mistral
	}

	option = {
		name = "Seek a fair compromise."
		# Balancing relations while maintaining Vale's stance
		add_political_power = -20
		improve_relations_with_Mistral = 0.5
		decrease_stability = 0.02

		trigger_event = MIS_mistral_cautious_approach

	}

	option = {
		name = "Issue a stern warning."
		# Aggressive diplomatic stance, risking further instability
		add_political_power = 30
		increase_stability = 0.10
		worsen_relations_with_Mistral = 1.0

		trigger_event = MIS_mistral_responds_to_threat

	}
}

	option = {

		create_non_aggression_pact = VAL

	}

	option = {
		name = "Reject the Offer"
		ai_chance = { factor = 10 }

		trigger_event = { id = VAL_vacuo_response_reject }
	}
}

	picture = GFX_report_event_generic_diplomacy

	is_triggered_only = yes
	trigger = {
		# Ensure this event is triggered after the focus completes
		has_focus = VAL_colonial_disputes
	}

	option = {
		name = "Calm the colonists and ease tensions."
		# Option for calming the colonists
		add_political_power = -50
		increase_stability = 0.05
		trigger_event = VAL_colonists_calmed
	}

	option = {
		name = "Support the colonists' demands."
		# Option for supporting colonists
		add_political_power = -30
		increase_recruitment = 0.10  # Increased recruitment from support
		trigger_event = VAL_colonists_supported
	}

	option = {
		name = "Ignore the colonists' complaints."
		# Option for ignoring the colonists
		add_political_power = 20
		increase_stability = -0.05  # A slight stability decrease due to ignoring the issue
		trigger_event = VAL_colonists_ignored
	}
}
