		# THIS is a target

		score = {
			base = 1
			# Add tension added by country
			modifier = {
				always = yes 
				add = has_added_tension_amount

			modifier = {
				always = yes

			}
			# add factories

				always = yes

				is_major = yes

				is_faction_leader = yes 
				add = 50

				NOT = { has_government = grimms_influence }
				factor = 0 
			}

				exists = no 
				factor = 0

local_grimm_country = {
	# Valid example of mission scorer:
	targets = {
		# describe the target to consider
		targets_dynamic = no
		target_non_existing = no
		#target_array = global.majors
		target_array = global.countries
		# MTTH like score description
		# THIS is a target
		# FROM is the initiator
		score = {
			base = 1
			
			modifier = {
				is_major = yes
				factor = 10
			}
			# Add tension added by country
			modifier = {
				always = yes 
				add = has_added_tension_amount
			}
			# Add total number of Divisions 
			modifier = {
				always = yes
				add = num_armies 
			}
			modifier = {
				tag = GRM # If the Dark Lands are still Grimm, they are something to be vary about - always 
				factor = 1.8
			}
			# If neighbor, more scary
			modifier = {
				is_neighbor_of = FROM
				factor = 2
			}
			# Same continent also more scary
			modifier = {
				OR = {
					AND = {
						THIS.capital_scope = { is_on_continent = sanus }
						FROM.capital_scope = { is_on_continent = sanus }
					}
					AND = {
						THIS.capital_scope = { is_on_continent = anima }
						FROM.capital_scope = { is_on_continent = anima }
					}
					AND = {
						THIS.capital_scope = { is_on_continent = solitas }
						FROM.capital_scope = { is_on_continent = solitas }
					}
					AND = {
						THIS.capital_scope = { is_on_continent = menagerie }
						FROM.capital_scope = { is_on_continent = menagerie }
					}
					AND = {
						THIS.capital_scope = { is_on_continent = draco }
						FROM.capital_scope = { is_on_continent = draco }
					}
				}
				factor = 3
			}
			modifier = {
				add = num_armies
			}
			# add factories
			modifier = {
				always = yes 
				add = num_of_factories
			}
			modifier = {
				NOT = { has_government = grimms_influence }
				factor = 0 
			}
		}
	}
}
